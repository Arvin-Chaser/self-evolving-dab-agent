# Helper: parse SQL INSERT tuples robustly
def parse_sql_insert_values(values_text):
    """
    Given the part after VALUES in an INSERT statement (possibly many tuples),
    return a list of tuple-content strings (without outer parentheses).
    """
    tuples = []
    i = 0
    n = len(values_text)
    while i < n:
        # find next '('
        while i < n and values_text[i] != '(':
            i += 1
        if i >= n:
            break
        depth = 0
        start = i
        while i < n:
            c = values_text[i]
            if c == '(':
                depth += 1
            elif c == ')':
                depth -= 1
                if depth == 0:
                    # include from start+1 to i-1
                    tuples.append(values_text[start+1:i])
                    i += 1
                    break
            i += 1
    return tuples

def split_sql_tuple_values(tuple_text):
    """
    Split a single tuple value string (contents inside parentheses) into a list of values,
    respecting single-quoted SQL literal quoting where '' is escaped quote.
    """
    vals = []
    i = 0
    n = len(tuple_text)
    cur = []
    in_quote = False
    while i < n:
        c = tuple_text[i]
        if in_quote:
            if c == "'":
                # check for doubled ''
                if i+1 < n and tuple_text[i+1] == "'":
                    cur.append("'")
                    i += 2
                    continue
                else:
                    in_quote = False
                    i += 1
                    continue
            else:
                cur.append(c)
                i += 1
                continue
        else:
            if c == "'":
                in_quote = True
                i += 1
                continue
            elif c == ",":
                vals.append("".join(cur).strip())
                cur = []
                i += 1
                continue
            else:
                cur.append(c)
                i += 1
                continue
    # append last
    if cur:
        vals.append("".join(cur).strip())
    # Normalize SQL NULL to Python None, unquote already handled
    norm = []
    for v in vals:
        if v.upper() == "NULL":
            norm.append(None)
        else:
            norm.append(v)
    return norm

def extract_year_from_text(text):
    if not text:
        return None
    # prefer four-digit year 1900-2099
    m = re.search(r'(19|20)\d{2}', text)
    if m:
        return int(m.group(0))
    return None

def parse_cpc_field(value):
    """
    Given a text field from publicationinfo that holds CPC info (may be JSON-like or raw),
    return a list of code strings found (normalized).
    Attempts:
      - json.loads if looks like JSON array/dict
      - regex find of "code": "..."
      - regex find of CPC-like tokens
    """
    if not value:
        return []
    val = value.strip()
    codes = []
    # Try JSON
    if (val.startswith('[') or val.startswith('{')):
        try:
            obj = json.loads(val)
            # obj could be list of dicts
            if isinstance(obj, list):
                for item in obj:
                    if isinstance(item, dict):
                        code = item.get("code") or item.get("symbol") or item.get("cpc")
                        if code:
                            codes.append(str(code).strip())
            elif isinstance(obj, dict):
                # maybe dict with keys
                if "code" in obj:
                    codes.append(str(obj["code"]).strip())
        except Exception:
            # fall through to regex heuristics
            pass
    # regex for explicit "code": "..."
    codes += re.findall(r'"code"\s*:\s*"([^"]+)"', val)
    codes += re.findall(r"'code'\s*:\s*'([^']+)'", val)
    # generic CPC pattern e.g., H01M10/0565 or C01B33/00 or B29L2031/3017
    generic = re.findall(r'\b[A-H][0-9]{1,2}[A-Z](?:[0-9A-Z])*\/\d+[0-9A-Z\/]*\b', val)
    codes += generic
    # Also catch patterns like A01B1/00 (no extra trailing letters)
    generic2 = re.findall(r'\b[A-H][0-9]{1,2}[A-Z]\d*(?:\/\d+)?\b', val)
    codes += generic2
    # normalize (unique, strip)
    norm = []
    for c in codes:
        cc = c.strip()
        if cc:
            norm.append(cc)
    # dedupe while preserving order
    seen = set()
    out = []
    for c in norm:
        if c not in seen:
            seen.add(c)
            out.append(c)
    return out

              1) If exact match in metadata and its level == TARGET_LEVEL -> return it.
              2) Else, find among metadata symbols those that are a prefix of raw_code (raw_code startswith symbol)

              3) Else, attempt progressive truncation of raw_code by removing trailing segments split by '/', then by digits,
                 and test for membership in metadata of level TARGET_LEVEL.
              4) If still not found, return None.
            Note: This uses metadata presence for mapping; string-prefix matching is used only to find a metadata symbol
            that matches the hierarchy as recorded in metadata.

                    cand = "/".join(parts[:k])
                    if cand in symbol_to_level and symbol_to_level.get(cand) == TARGET_LEVEL:

                parts.pop()  # remove last part and try again
            # as last resort, attempt to find any metadata symbol that is prefix of rc regardless of level,
            # then climb up (find its ancestors in metadata that have level TARGET_LEVEL) by prefixing.

                    while segs:
                        cand = "/".join(segs)
                        if cand in symbol_to_level and symbol_to_level.get(cand) == TARGET_LEVEL:
                            candidate2 = cand
                            break
                        segs.pop()
                    if candidate2:
                        return candidate2
            # Not found
            return None

            # build series in order full_years
            series = [year_counts.get(y, 0) for y in full_years]
            # apply EMA: initialize EMA_first = value_first_year
            ema_vals = []
            if not series:
                continue
            ema = series[0]
            ema_vals.append(ema)
            for v in series[1:]:
                ema = ALPHA * v + (1 - ALPHA) * ema
                ema_vals.append(ema)
            # map EMA values back to years
            year_to_ema = {yr: e for yr, e in zip(full_years, ema_vals)}
            # select best year: max EMA, on ties choose latest year (deterministic)
            max_ema = None
            best_year = None
            for yr in full_years:
                val = year_to_ema[yr]
                if (max_ema is None) or (val > max_ema) or (val == max_ema and yr > best_year):
                    max_ema = val
                    best_year = yr
            ema_by_code[code] = year_to_ema
            best_year_by_code[code] = best_year
        add_stage("S7_transformation_calculation", "completed",
                  "Computed EMA series and best years for each canonical code.",
                  outputs=[
                      "EMA computed for {} codes".format(len(ema_by_code)),
                      "Target smoothing alpha: {}".format(ALPHA),
                      "Full years range: {}-{}".format(y_min, y_max)
                  ])
    except Exception as e:
        add_stage("S7_transformation_calculation", "failed", "Error during parsing/aggregation/EMA computation.", error=str(e))
        raise

    # Stage S8: Answer synthesis and validation
    try:
        add_stage("S8_answer_synthesis_validation", "started",
                  "Filter codes to level 5 with best year == 2022, validate and write final answer.")
        # Filter codes whose best year == TARGET_YEAR and whose level is TARGET_LEVEL
        final_codes = []
        for code, by in best_year_by_code.items():
            lvl = symbol_to_level.get(code)
            if lvl == TARGET_LEVEL and by == TARGET_YEAR:
                final_codes.append(code)
        # Deduplicate and sort ascending
        final_codes = sorted(set(final_codes))
        # Validation checks
        total_canonical_codes_considered = len(ema_by_code)
        codes_with_best_target = sum(1 for c, y in best_year_by_code.items() if y == TARGET_YEAR)
        outputs_summary = [
            "Total canonical codes considered: {}".format(total_canonical_codes_considered),
            "Codes with best year == {}: {}".format(TARGET_YEAR, codes_with_best_target),
            "Final output codes (level {} with best year {}): {}".format(TARGET_LEVEL, TARGET_YEAR, len(final_codes))
        ]
        # Ensure each output code exists in metadata and is level TARGET_LEVEL
        invalid_codes = [c for c in final_codes if c not in symbol_to_level or symbol_to_level.get(c) != TARGET_LEVEL]
        if invalid_codes:
            raise RuntimeError("Validation failed: some output codes not in metadata or not at level {}: {}".format(TARGET_LEVEL, invalid_codes))

        # Write final answers to FINAL_ANSWER_PATH, one code per line
        Path(FINAL_ANSWER_PATH).parent.mkdir(parents=True, exist_ok=True)
        with open(FINAL_ANSWER_PATH, "w", encoding="utf-8") as fa:
            for code in final_codes:
                fa.write(code.strip() + "\n")
        print("FINAL: Wrote {} codes to {}".format(len(final_codes), FINAL_ANSWER_PATH))

        add_stage("S8_answer_synthesis_validation", "completed",
                  "Final answer written and validated.",
                  evidence_used=[FINAL_ANSWER_PATH, CPC_DEFINITION_SQL_PATH, PUBLICATION_DB_PATH],
                  outputs=outputs_summary + ["Final codes count: {}".format(len(final_codes))],
                  assumptions=["Output ordering: sorted ascending; duplicates removed."],
                  uncertainties=["Mapping coverage limited by metadata parsing heuristics; some codes may be unmapped."])
    except Exception as e:
        add_stage("S8_answer_synthesis_validation", "failed", "Failed during final answer synthesis/validation.", error=str(e))
        raise

if __name__ == "__main__":
    try:
        main()
    except Exception as e:
        # Ensure we capture exception in trace and still write
        tb = traceback.format_exc()
        add_stage("S8_answer_synthesis_validation", "failed", "Unhandled exception during execution.", error=tb)
        print("ERROR: Unhandled exception:", file=sys.stderr)
        print(tb, file=sys.stderr)
        # Attempt to still write trace (already done in add_stage)
        sys.exit(1)
