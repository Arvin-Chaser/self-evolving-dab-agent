# Helper: parse SQL CREATE TABLE to get column order for cpc_definition
def parse_create_table_columns(sql_text, table_name="cpc_definition"):
    pattern = re.compile(r'CREATE\s+TABLE\s+["`]?%s["`]?\s*\((.*?)\);' % re.escape(table_name), re.IGNORECASE | re.DOTALL)
    m = pattern.search(sql_text)
    if not m:
        return None
    cols_blob = m.group(1)
    cols = []
    # Split by commas but naive split may break on constraints; instead find column names by regex
    # Capture patterns like "  `symbol` TEXT," or '"symbol" TEXT,' or 'symbol TEXT,'
    for line in cols_blob.splitlines():
        line = line.strip().rstrip(',')
        if not line:
            continue
        cm = re.match(r'["`]?([A-Za-z0-9_]+)["`]?\s+.*', line)
        if cm:
            cols.append(cm.group(1))
    return cols

# Helper: parse INSERT INTO cpc_definition values
def extract_tuples_from_insert_values(values_blob):
    # values_blob contains content after VALUES up to the semicolon, e.g. "(...),(...),(...)".
    tuples = []
    i = 0
    n = len(values_blob)
    while i < n:
        # find '('
        while i < n and values_blob[i] != '(':
            i += 1
        if i >= n:
            break
        i += 1  # skip '('
        cur = []
        token = None
        # We will parse expecting SQL strings quoted by single quotes, handling doubled single quotes.
        while i < n:
            ch = values_blob[i]
            if ch == "'":
                # parse string token
                i += 1
                sb = []
                while i < n:
                    if values_blob[i] == "'":
                        # check if next is also a single quote -> escaped quote
                        if i + 1 < n and values_blob[i+1] == "'":
                            sb.append("'")
                            i += 2
                            continue
                        else:
                            i += 1
                            break
                    else:
                        sb.append(values_blob[i])
                        i += 1
                cur.append(''.join(sb))
                # after string token, skip spaces and possible comma or closing paren
                while i < n and values_blob[i] in " \t\r\n":
                    i += 1
                if i < n and values_blob[i] == ',':
                    i += 1
                    continue
                elif i < n and values_blob[i] == ')':
                    i += 1
                    break
                else:
                    # Could be next is another token (like NULL or number). We'll handle non-quoted tokens:
                    while i < n and values_blob[i] in " \t\r\n":
                        i += 1
                    if i < n and values_blob[i] not in ",)":
                        # parse unquoted token
                        start = i
                        while i < n and values_blob[i] not in ",)":
                            i += 1
                        cur.append(values_blob[start:i].strip())
                        if i < n and values_blob[i] == ',':
                            i += 1
                            continue
                        elif i < n and values_blob[i] == ')':
                            i += 1
                            break
                    elif i < n and values_blob[i] == ',':
                        i += 1
                        continue
                    elif i < n and values_blob[i] == ')':
                        i += 1
                        break
            elif ch == ')':
                i += 1
                break
            else:
                # non-quoted token: read until comma or )
                start = i
                while i < n and values_blob[i] not in ",)":
                    i += 1
                cur.append(values_blob[start:i].strip())
                if i < n and values_blob[i] == ',':
                    i += 1
                    continue
                elif i < n and values_blob[i] == ')':
                    i += 1
                    break
        tuples.append(cur)
    return tuples

def parse_cpc_definition(sql_path):
    """
    Parse CPC definition SQL to extract mapping:
    - columns order (inferred from CREATE TABLE)
    - from INSERT INTO cpc_definition get tuples mapped to columns
    Returns:
      columns (list) or None, entries (list of dicts)
    """
    try:
        text = open(sql_path, "r", encoding="utf-8", errors="ignore").read()
    except Exception as e:
        raise

    cols = parse_create_table_columns(text, "cpc_definition")
    entries = []
    # Find all INSERT INTO cpc_definition ... VALUES ...;
    insert_pattern = re.compile(r'INSERT\s+INTO\s+["`]?cpc_definition["`]?(?:\s*\([^\)]*\))?\s+VALUES\s*(.*?);', re.IGNORECASE | re.DOTALL)
    matches = list(insert_pattern.finditer(text))
    for m in matches:
        values_blob = m.group(1)
        tuples = extract_tuples_from_insert_values(values_blob)
        for t in tuples:
            # map to columns if available
            if cols and len(t) <= len(cols):
                d = {}
                for idx, val in enumerate(t):
                    d[cols[idx]] = val
                entries.append(d)
            else:
                # if no cols, store as list with numeric keys
                d = {"_col_%d" % i: v for i, v in enumerate(t)}
                entries.append(d)
    return cols, entries

# Robust extractor for CPC codes from raw text/json
def extract_codes_from_raw(raw):
    codes = []
    if raw is None:
        return codes
    s = raw.strip()
    # Try JSON parse
    try:
        parsed = json.loads(s)
        if isinstance(parsed, list):
            for item in parsed:
                if isinstance(item, dict):
                    # look for 'code' or 'symbol'
                    code = item.get("code") or item.get("symbol") or item.get("cpc")
                    if code:
                        codes.append(code.strip())
                elif isinstance(item, str):
                    codes.append(item.strip())
        elif isinstance(parsed, dict):
            # maybe dict of codes
            for k, v in parsed.items():
                if isinstance(v, str) and re.search(r'[A-H]\d', v):
                    codes.append(v.strip())
    except Exception:
        # Not JSON; try to find typical CPC code patterns via regex
        pass

    if not codes:
        # regex to find patterns like A01B33/00 or B32B2262/101 or H01M10/0565 etc.
        # pattern: letter (A-H), 1-2 digits, uppercase letter, optional digits, slash, digits (and optional further digits)
        # Also capture patterns like C08J5/246 (C08J5/246) where there's an extra digit chunk after letter
        regex = re.compile(r'\b[A-H][0-9]{1,2}[A-Z][0-9A-Z]*/[0-9A-Za-z]+(?:/[0-9A-Za-z]+)?\b')
        found = regex.findall(s)
        # fallback broader pattern for cases like B29L2031/3017
        if not found:
            regex2 = re.compile(r'\b[A-H][0-9]{1,2}[A-Z][0-9A-Za-z]*/[0-9A-Za-z/]+\b')
            found = regex2.findall(s)
        # also capture simple patterns like 'F16H47/04' etc.
        for f in found:
            codes.append(f)
    # Deduplicate while preserving order
    seen = set()
    out = []
    for c in codes:
        if c is None:
            continue
        c2 = c.strip()
        if c2 and c2 not in seen:
            seen.add(c2)
            out.append(c2)
    return out

# Extract year from a natural-language date string
def extract_year_from_text(s):
    if s is None:
        return None
    s = s.strip()
    # look for 4-digit year between 1900 and 2099
    m = re.search(r'(19|20)\d{2}', s)
    if m:
        return int(m.group(0))
    # fallback: look for last two digits patterns like '21' ambiguous - skip
    return None

def safe_open_db(path):
    if not os.path.exists(path):
        raise FileNotFoundError(f"Database file not found: {path}")
    conn = sqlite3.connect(path)
    conn.row_factory = sqlite3.Row
    return conn

                    best_years[code] = {"best_year": best_y, "best_ema": max_ema}
            add_stage(
                stage="S7_transformation_calculation",
                status="completed",
                summary="Computed EMA per symbol and identified best year per symbol.",
                evidence_used=[f"Years range: {min_year if counts else 'N/A'}..{max_year if counts else 'N/A'}", f"Sample EMA entry: {list(ema_results.items())[:3]}"],
                assumptions=["EMA alpha=0.2; EMA initialized to first year's count."],
                outputs={"num_codes_with_ema": len(ema_results), "num_best_years": len(best_years)},
                uncertainties=["If counts are zero across all years for a code, EMA will be zero; tie-break: choose latest year."],
                error=None
            )
        except Exception as e:
            add_stage(
                stage="S7_transformation_calculation",
                status="failed",
                summary="Failed while computing EMA or best years.",
                evidence_used=[f"Counts sample: {list(counts.items())[:10]}"],
                assumptions=[],
                outputs=[],
                uncertainties=[],
                error=traceback.format_exc()
            )
            write_trace()
            raise

        # S8 Answer synthesis and validation
        try:
            add_stage(
                stage="S8_answer_synthesis_validation",
                status="started",
                summary="Filter symbols whose best year is 2022 and validate outputs before writing final answer.",
                evidence_used=[f"Best years sample: {list(best_years.items())[:10]}"],
                assumptions=["Only symbols explicitly mapped to level 5 in metadata are considered valid outputs."],
                outputs=[],
                uncertainties=[],
                error=None
            )
            # Collect codes whose best year == 2022
            target_year = 2022
            final_codes = []
            # Only include codes that are in symbol_to_level and level==5 (we ensured counts only included such codes)
            for code, info in best_years.items():
                if info["best_year"] == target_year:
                    # validate membership in metadata and level
                    level = symbol_to_level.get(code)
                    if level != 5:
                        # skip: does not match required level
                        continue
                    final_codes.append(code)
            # Deduplicate and sort ascending deterministically
            final_codes = sorted(set(final_codes))
            # Validation checks
            validation = {
                "total_canonical_codes_considered": len(ema_results),
                "num_codes_best_year_2022": sum(1 for v in best_years.values() if v.get("best_year") == target_year),
                "num_codes_output": len(final_codes),
                "all_codes_unique": len(final_codes) == len(set(final_codes))
            }
            # Additional check: all output codes present in metadata and at level 5
            invalid_codes = [c for c in final_codes if (c not in symbol_to_level) or (symbol_to_level.get(c) != 5)]
            if invalid_codes:
                # This should not happen given prior filtering; if it does, remove them
                final_codes = [c for c in final_codes if c not in invalid_codes]
            # Print stats
            print("Validation stats:", validation)
            if len(final_codes) == 0:
                print("Warning: No CPC level-5 codes found whose best EMA year is 2022.")
            # Write final answer file: one code per line
            os.makedirs(os.path.dirname(FINAL_ANSWER_PATH), exist_ok=True)
            with open(FINAL_ANSWER_PATH, "w", encoding="utf-8") as fout:
                for c in final_codes:
                    fout.write(c.strip() + "\n")
            add_stage(
                stage="S8_answer_synthesis_validation",
                status="completed",
                summary="Wrote final answer file with CPC codes whose best year is 2022.",
                evidence_used=[f"Final codes count: {len(final_codes)}", f"FINAL_ANSWER_PATH: {FINAL_ANSWER_PATH}"],
                assumptions=[],
                outputs={"final_codes_count": len(final_codes), "final_codes_sample": final_codes[:20]},
                uncertainties=[],
                error=None
            )
            print(f"Wrote final answer to {FINAL_ANSWER_PATH} with {len(final_codes)} codes.")
        except Exception as e:
            add_stage(
                stage="S8_answer_synthesis_validation",
                status="failed",
                summary="Failed during answer finalization or validation.",
                evidence_used=[],
                assumptions=[],
                outputs=[],
                uncertainties=[],
                error=traceback.format_exc()
            )
            write_trace()
            raise

    except Exception as e_main:
        # Ensure trace is written and print error
        print("An error occurred during processing:", file=sys.stderr)
        traceback.print_exc()
        # If not already recorded, add a failure stage
        add_stage(
            stage="S8_answer_synthesis_validation" if True else "unknown",
            status="failed",
            summary="Unhandled exception in main execution.",
            evidence_used=[],
            assumptions=[],
            outputs=[],
            uncertainties=[],
            error=traceback.format_exc()
        )
        write_trace()
        sys.exit(1)

if __name__ == "__main__":
    main()
